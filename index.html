How to set up the database in Supabase (SQL Script)
-- 1. Create the table in your Supabase SQL Editor CREATE TABLE disciplinary_cases ( id TEXT PRIMARY KEY, emp TEXT NOT NULL, dept TEXT, reported DATE, type TEXT, "desc" TEXT, owner TEXT, stage TEXT, s_date DATE, sla INT, comp DATE, reason TEXT, created_at TIMESTAMP WITH TIME ZONE DEFAULT timezone('utc'::text, now()) NOT NULL ); -- 2. Enable public read and write access ALTER TABLE disciplinary_cases ENABLE ROW LEVEL SECURITY; CREATE POLICY "Public Access Policy" ON disciplinary_cases AS PERMISSIVE FOR ALL TO public USING (true) WITH CHECK (true);
